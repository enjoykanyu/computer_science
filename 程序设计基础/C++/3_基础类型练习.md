### 算数运算

#include <iostream>
```cpp
int main () {
int a = 4;
int b = 3;

a = a + b;
std::cout << "a: " << a << '\n';
a += b;    
std::cout << "a: " << a << '\n';

a = a - b;
std::cout << "a: " << a << '\n';
a -= b;    
std::cout << "a: " << a << '\n';

a = a * b;
std::cout << "a: " << a << '\n';
a *= b;    
std::cout << "a: " << a << '\n';

a = a / b;
std::cout << "a: " << a << '\n';
a /= b;    
std::cout << "a: " << a << '\n';

a = a % b;
std::cout << "a: " << a << '\n';
}
```
